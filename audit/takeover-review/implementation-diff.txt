+export function describeTakeoverAccess(conversation, actor, { enabled = isManualTakeoverEnabled() } = {}) {
+  const permitted = canTakeoverConversation(actor?.role);
+  const botPaused = conversation?.status === 'assigned' || conversation?.status === 'paused';
+  const base = {
+    canTakeover: false,
+    takeoverBlock: null,
+    reentryStatus: reentryStatusOf(conversation),
+    botPaused,
+    statusVersion: conversation?.statusVersion ?? null,
+    canReassign: Boolean(
+      canReassignConversation(actor?.role)
+      && botPaused
+      && conversation?.assignedAgentId
+      && !agentsMatch(conversation.assignedAgentId, actor?.agentId),
+    ),
+  };

+  if (!enabled) return { ...base, takeoverBlock: 'disabled' };
+  if (!permitted) return { ...base, takeoverBlock: 'forbidden' };
+  if (!conversation) return { ...base, takeoverBlock: 'not_found' };
+  if (conversation.status === 'closed') return { ...base, takeoverBlock: 'closed' };
+
+  const hasOwner = Boolean(conversation.assignedAgentId)
+    && (conversation.status === 'assigned' || conversation.status === 'paused');
+  if (hasOwner && agentsMatch(conversation.assignedAgentId, actor?.agentId)) {
+    return { ...base, takeoverBlock: 'already_owner' };
+  }
+  if (hasOwner) return { ...base, takeoverBlock: 'owned_by_other' };
+
+  if (
+    conversation.status === 'bot'
+    || conversation.status === 'waiting_human'
+    || conversation.status === 'paused'
+  ) {
+    return { ...base, canTakeover: true, takeoverBlock: null };
+  }

+import conversationRepository from '../repositories/conversation.repository.js';
+import { isConversationDbEnabled, withTransaction } from '../db/index.js';
+
+const CONVERSATION_ID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
+
+/**
+ * Después del motor y antes de persistir o enviar una respuesta automática.
+ * Si la toma ya commiteó, este turno no publica menú ni fallback.
+ */
+export async function evaluateAutomaticReply(conversation, statusVersionAtStart, deps = {}) {
+  if (!conversation?.id) return { suppress: false, reason: 'no_conversation' };
+  if (!CONVERSATION_ID.test(String(conversation.id))) {
+    return { suppress: false, reason: 'not_persisted_id' };
+  }
+
+  const dbEnabled = deps.dbEnabled || isConversationDbEnabled;
+  if (!dbEnabled()) return { suppress: false, reason: 'db_disabled' };
+
+  const repo = deps.conversationRepo || conversationRepository;
+  const runTransaction = deps.runTransaction || withTransaction;
+
+  return runTransaction(async (transaction) => {
+    const locked = await repo.findByIdForUpdate(conversation.id, { transaction });
+    if (!locked) return { suppress: false, reason: 'missing_row' };
+    if (locked.status !== 'bot') {
+      return { suppress: true, reason: 'not_bot', conversation: locked };
+    }
+    const start = Number(statusVersionAtStart);
+    const current = Number(locked.statusVersion);
+    if (Number.isInteger(start) && Number.isInteger(current) && start !== current) {
+      return { suppress: true, reason: 'version', conversation: locked };
+    }
+    return { suppress: false, reason: 'current', conversation: locked };
+  });
+}
+
+export const conversationTurnGuard = {
+  suppressAutomaticReply(conversation, statusVersionAtStart) {
+    return evaluateAutomaticReply(conversation, statusVersionAtStart);
+  },
+};
