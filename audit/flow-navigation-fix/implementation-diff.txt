+import { FlowFieldValidationError } from './flow-field-validation.js';
+
+const KEYCAP = /([0-9])\uFE0F?\u20E3/g;
+const COMBINING_MARKS = /[\u0300-\u036f]/g;
+
+/**
+ * Forma canónica para comparar entradas con opciones.
+ * No modifica el texto que se persiste.
+ * NFKC, keycaps a dígitos, sin diacríticos, minúsculas y espacios colapsados.
+ * @param {unknown} value
+ * @returns {string}
+ */
+export function normalizeFlowMatchText(value) {
+  if (typeof value === 'number' || typeof value === 'boolean') {
+    return normalizeFlowMatchText(String(value));
+  }
+  if (typeof value !== 'string') return '';
+  const canonical = value
+    .normalize('NFKC')
+    .replace(KEYCAP, '$1')
+    .normalize('NFD')
+    .replace(COMBINING_MARKS, '')
+    .toLowerCase()
+    .trim()
+    .replace(/\s+/g, ' ');
+  return canonical;
+}
+
+/**
+ * @param {unknown} value
+ * @param {{ strict?: boolean, flowKey?: string, version?: string, nodeId?: string, path?: string }} [ctx]
+ * @returns {string}
+ */
+export function transitionMatchKey(value, ctx = {}) {
+  if (typeof value === 'number' || typeof value === 'boolean') {
+    return normalizeFlowMatchText(String(value));
+  }
+  if (typeof value === 'string') return normalizeFlowMatchText(value);
+  if (value == null || value === '') return '';
+  if (ctx.strict) {
+    throw new FlowFieldValidationError({
+      flowKey: ctx.flowKey || 'unknown',
+      version: ctx.version,
+      nodeId: ctx.nodeId,
+      path: ctx.path || 'transition.value',
+      expectedType: 'string',
+      value,
+    });
+  }
+  return '';
+}
+
+/**
+ * El needle tiene que aparecer como tokens completos, no dentro de otra palabra.
+ * @param {string} input
+ * @param {string} needle
+ * @returns {boolean}
+ */
+export function includesAsTokens(input, needle) {
+  if (!input || !needle) return false;
+  const escaped = needle.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
+  return new RegExp(`(?:^|\\s)${escaped}(?:\\s|$)`).test(input);
+}

+import { normalizeFlowMatchText } from './flow-match-text.js';
+
+const OPTION_LINE = /(?:^|\n)[ \t]*((?:[0-9]\uFE0F?\u20E3)|[0-9]{1,2})[ \t]+\*([^*\n]+)\*/g;
+
+/**
+ * Opciones numeradas visibles en el mensaje del nodo.
+ * @param {string} message
+ * @returns {{ digit: string, label: string }[]}
+ */
+export function parseEnumeratedMenuOptions(message) {
+  if (typeof message !== 'string' || !message) return [];
+  const options = [];
+  const pattern = new RegExp(OPTION_LINE.source, 'g');
+  let match = pattern.exec(message);
+  while (match) {
+    const digit = match[1].replace(/\uFE0F?\u20E3/g, '');
+    const label = normalizeFlowMatchText(match[2]);
+    if (digit && label) options.push({ digit, label });
+    match = pattern.exec(message);
+  }
+  return options;
+}
+
+/**
+ * @param {object | null | undefined} node
+ * @param {Map<string, object>} exact
+ */
+export function indexEnumeratedLabels(node, exact) {
+  const byLabel = new Map();
+  const ambiguousLabels = new Set();
+  for (const option of parseEnumeratedMenuOptions(node?.message)) {
+    const transition = exact.get(option.digit);
+    if (!transition || transition.type !== 'match' || transition.nextNode == null) continue;
+    if (ambiguousLabels.has(option.label)) continue;
+    const current = byLabel.get(option.label);
+    if (!current) {
+      byLabel.set(option.label, { ...option, transition });
+      continue;
+    }
+    if (current.transition.nextNode !== transition.nextNode) {
+      byLabel.delete(option.label);
+      ambiguousLabels.add(option.label);
+    }
+  }
+  return { byLabel, ambiguousLabels };
+}

+ * @param {number} count
+ * @param {boolean} humanAvailable
+ * @returns {string}
+ */
+export function formatUnmatchedMenuReply(node, count, humanAvailable) {
+  const menu = String(node?.message || '').trim();

+import { includesAsTokens, transitionMatchKey } from './flow-match-text.js';
+import { indexEnumeratedLabels } from './flow-menu-options.js';
+
+const MIN_INCLUDE_LENGTH = 3;
+
+function addExact(exact, ambiguousExact, key, transition) {
+  if (!key) return;
+  if (ambiguousExact.has(key)) return;
+  const current = exact.get(key);
+  if (!current) {
+    exact.set(key, transition);
+    return;
+  }
+  if (current.nextNode === transition.nextNode) return;
+  exact.delete(key);
+  ambiguousExact.add(key);
+}
+
+/**
+ * Índice de matching de un nodo. La misma función la usan el compile y el motor.
+ * @param {object} node
+ * @param {{ strict?: boolean, flowKey?: string, version?: string }} [ctx]
+ */
+export function buildNodeTransitionIndex(node, ctx = {}) {
+  const exact = new Map();
+  const ambiguousExact = new Set();
+  const includes = [];
+  let defaultTransition = null;
+  const transitions = Array.isArray(node?.transitions) ? node.transitions : [];
+  const baseCtx = {
+    strict: Boolean(ctx.strict),
+    flowKey: ctx.flowKey,
+    version: ctx.version,
+    nodeId: node?.id,
+  };
+
+  for (const transition of transitions) {
+    const type = transition?.type;
+    if (type === 'match') {
+      addExact(
+        exact,
+        ambiguousExact,
+        transitionMatchKey(transition.value, {
+          ...baseCtx,
+          path: `nodes.${node?.id}.transitions[].value`,
+        }),
+        transition,
+      );
+      continue;
+    }
+    if (type === 'matchAny' && Array.isArray(transition.value)) {
+      for (const rawValue of transition.value) {
+        addExact(
+          exact,
+          ambiguousExact,
+          transitionMatchKey(rawValue, {
+            ...baseCtx,
+            path: `nodes.${node?.id}.transitions[].value[]`,
+          }),
+          transition,
+        );
+      }
+      continue;
+    }
+    if (type === 'matchIncludes') {
+      const needle = transitionMatchKey(transition.value, {
+        ...baseCtx,
+        path: `nodes.${node?.id}.transitions[].value`,
+      });
+      if (needle.length >= MIN_INCLUDE_LENGTH) {
+        includes.push({ needle, transition });
+      }
+      continue;
+    }
+    if (type === 'default' || transition?.default === true) {
+      if (!defaultTransition) defaultTransition = transition;
+    }
+  }
+
+  return {
+    exact,
+    ambiguousExact,
+    includes,
+    defaultTransition,
+    enumerated: indexEnumeratedLabels(node, exact),
+  };
+}
+
+/**
+ * @param {object | null | undefined} node
+ * @param {ReturnType<typeof buildNodeTransitionIndex> | null} index
+ * @param {string} input texto ya normalizado
+ * @param {string | null | undefined} fallbackNodeId
+ */
+export function resolveIndexedTransition(node, index, input, fallbackNodeId) {
+  if (!node) {
+    return {
+      nextNodeId: fallbackNodeId,
+      reason: 'node_missing',
+      usedFallback: true,
+      matchedTransition: null,
+    };
+  }
+  if (!node.transitions || node.transitions.length === 0) {
+    return {
+      nextNodeId: node.nextNode || fallbackNodeId,
+      reason: node.nextNode ? 'node_next' : 'fallback',
+      usedFallback: !node.nextNode,
+      matchedTransition: null,
+    };
+  }
+
+  if (index.ambiguousExact.has(input)) {
+    return stay(node, 'ambiguous_exact');
+  }
+  const exact = index.exact.get(input);
+  if (exact) {
+    const reason = exact.type === 'matchAny' ? 'matchAny' : 'match';
+    return {
+      nextNodeId: exact.nextNode,
+      reason,
+      usedFallback: false,
+      matchedTransition: exact,
+    };
+  }
+  if (index.enumerated.ambiguousLabels.has(input)) {
+    return stay(node, 'ambiguous_label');
+  }
+  const labeled = index.enumerated.byLabel.get(input);
+  if (labeled) {
+    return {
+      nextNodeId: labeled.transition.nextNode,
+      reason: 'enumerated_label',
+      usedFallback: false,
+      matchedTransition: labeled.transition,
+    };
+  }
+
+  const includeHits = index.includes.filter((rule) => includesAsTokens(input, rule.needle));
+  const destinations = new Set(includeHits.map((hit) => hit.transition.nextNode));
+  if (destinations.size > 1) return stay(node, 'ambiguous_include');
+  if (destinations.size === 1) {
+    return {
+      nextNodeId: includeHits[0].transition.nextNode,
+      reason: 'matchIncludes',
+      usedFallback: false,
+      matchedTransition: includeHits[0].transition,
+    };
+  }
+
+  if (index.defaultTransition) {
+    return {
+      nextNodeId: index.defaultTransition.nextNode,
+      reason: 'default',
+      usedFallback: false,
+      matchedTransition: index.defaultTransition,
+    };
+  }
+
+  return {
+    nextNodeId: fallbackNodeId || node.id,
+    reason: fallbackNodeId ? 'fallback' : 'stay_on_node',
+    usedFallback: Boolean(fallbackNodeId),
+    matchedTransition: null,
+  };
+}
+
+function stay(node, reason) {
+  return {
+    nextNodeId: node.id,
+    reason,
+    usedFallback: false,
+    matchedTransition: null,
+  };
+}
